    <script>
        "use strict";

        const htmlEditor = document.getElementById("htmlCode");
        const cssEditor = document.getElementById("cssCode");
        const preview = document.getElementById("preview");

        function runCode() {
            const html = htmlEditor.value;
            const css = cssEditor.value;

            // Escape closing style tags so CSS stays inside its style element.
            const safeCSS = css.replace(/<\/style/gi, "<\\/style");

            const documentContent = [
                "<!DOCTYPE html>",
                '<html lang="en">',
                "<head>",
                '<meta charset="UTF-8">',
                '<meta name="viewport" content="width=device-width, initial-scale=1.0">',
                "<style>",
                safeCSS,
                "</style>",
                "</head>",
                "<body>",
                html,
                "</body>",
                "</html>"
            ].join("\n");

            // The preview is sandboxed: scripts and other active content are disabled.
            preview.srcdoc = documentContent;
        }

        function loadHTMLExample() {
            htmlEditor.value = `<h1>My First Website</h1>
<p>I am learning HTML with StudyNova.</p>
<h2>My Hobbies</h2>
<ul>
    <li>Reading</li>
    <li>Learning coding</li>
    <li>Creating websites</li>
</ul>
<button>Explore More</button>`;

            runCode();
        }

        function loadCSSExample() {
            cssEditor.value = `body {
    font-family: Arial, sans-serif;
    background: #fef3c7;
    padding: 25px;
}

h1 {
    color: #b45309;
    text-align: center;
}

h2 {
    color: #92400e;
}

p, li {
    font-size: 18px;
    line-height: 1.8;
}

button {
    background: #b45309;
    color: white;
    padding: 12px 20px;
    border: none;
    border-radius: 8px;
    cursor: pointer;
}`;

            runCode();
        }

        function resetCode() {
            const confirmed = confirm(
                "Reset both editors to the original example?"
            );

            if (confirmed) {
                htmlEditor.value = `<h1>Hello, StudyNova!</h1>
<p>Welcome to my first webpage.</p>
<button>Click Me</button>`;

                cssEditor.value = `body {
    font-family: Arial, sans-serif;
    background: #eef2ff;
    padding: 25px;
}

h1 {
    color: #4f46e5;
}

p {
    color: #374151;
    font-size: 18px;
}

button {
    background: #4f46e5;
    color: white;
    padding: 10px 18px;
    border: none;
    border-radius: 8px;
}`;

                runCode();
            }
        }

        document.getElementById("runBtn")
            .addEventListener("click", runCode);

        document.getElementById("htmlExampleBtn")
            .addEventListener("click", loadHTMLExample);

        document.getElementById("cssExampleBtn")
            .addEventListener("click", loadCSSExample);

        document.getElementById("resetBtn")
            .addEventListener("click", resetCode);

        // Display the starter example when the page opens.
        runCode();
    </script>
